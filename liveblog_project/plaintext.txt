// static/js/comments.js
(function () {
  'use strict';

  /* ===============================
     CSRF
  =============================== */
  function getCookie(name) {
    return document.cookie
      .split('; ')
      .find(c => c.startsWith(name + '='))
      ?.split('=')[1];
  }

  const CSRF = getCookie('csrftoken');

  async function parseJsonSafe(resp) {
    try {
      return await resp.json();
    } catch {
      return null;
    }
  }

  /* ===============================
     LISTING COMMENTS SYNC
  =============================== */
  function updateListingComments(itemId, count) {
    if (!itemId) return;

    let changes = {};
    try {
      changes = JSON.parse(sessionStorage.getItem('listing_changes') || '{}');
    } catch {
      changes = {};
    }

    if (!changes[itemId]) changes[itemId] = {};
    changes[itemId].comments_count = count;

    sessionStorage.setItem('listing_changes', JSON.stringify(changes));
  }

  /* ===============================
     UI HELPERS
  =============================== */
  function updateDetailCounter(count) {
    const el = document.getElementById('commentsCount');
    if (el) el.textContent = count;
  }

  function updateCardCounter(itemId, count) {
    const el = document.getElementById('comments-count-' + itemId);
    if (el) el.textContent = count;
  }

  function updateCommentsHeader(count) {
    const header = document.getElementById('commentsHeader');
    if (!header) return;

    header.innerHTML = Number(count) > 0
      ? `<h5 class="pt-5 mb-3 text-muted">Comments</h5>`
      : `<h5 class="pt-5 mb-3 text-muted">There are not comments yet.</h5>`;
  }

  function showFieldError(textarea, message) {
    if (!textarea) return;

    const form = textarea.closest('form');
    if (!form) return;

    form.querySelector('.field-error')?.remove();

    if (!message) return;

    const err = document.createElement('div');
    err.className = 'field-error';
    err.textContent = message;

    (textarea.closest('.form-floating') || textarea).after(err);
    textarea.focus();
  }

  function clearFieldError(textarea) {
    textarea?.closest('form')?.querySelector('.field-error')?.remove();
  }

  /* ===============================
     ADD COMMENT
  =============================== */
    function initClearCommentButton(textarea, clearBtn) {
    if (!textarea || !clearBtn) return;

    const toggle = () => {
      if (textarea.value.trim().length > 0) {
        clearBtn.classList.remove('hidden');
      } else {
        clearBtn.classList.add('hidden');
      }
    };

    // показываем / скрываем при вводе
    textarea.addEventListener('input', toggle);

    // очистка по клику
    clearBtn.addEventListener('click', () => {
      textarea.value = '';
      textarea.style.height = 'auto';
      toggle();
      textarea.focus();
    });

    // начальное состояние
    toggle();
  }
  function initAddComment() {
    const form = document.getElementById('commentForm');
    if (!form || form.__bound) return;
    form.__bound = true;

    const btn = document.getElementById('submitCommentBtn');
    const textarea = form.querySelector('textarea[name="text"]');
    const commentsList = document.getElementById('commentsList');
    const itemId = form.dataset.itemId;
    const clearBtn = document.getElementById('clearComment');
    initClearCommentButton(textarea, clearBtn);
    textarea.addEventListener('input', () => {
      clearFieldError(textarea);
    });

    btn.addEventListener('click', async (e) => {
      e.preventDefault();
      if (btn.disabled) return;

      const text = textarea.value.trim();
      if (!text) {
        showFieldError(textarea, 'Please write comment *');
        return;
      }

      btn.disabled = true;

      try {
        const resp = await fetch(form.action, {
          method: 'POST',
          headers: {
            'X-CSRFToken': CSRF,
            'X-Requested-With': 'XMLHttpRequest',
            'Accept': 'application/json'
          },
          body: new FormData(form)
        });

        const data = await parseJsonSafe(resp);

        if (resp.ok && data?.success) {
          commentsList?.insertAdjacentHTML('afterbegin', data.comment_html);

          textarea.value = '';
          textarea.style.height = 'auto';

          const count = data.comments_count;

          updateDetailCounter(count);
          updateCardCounter(itemId, count);
          updateListingComments(itemId, count);
          updateCommentsHeader(count);

          return;
        }

        showFieldError(textarea, data?.error || 'Server error');

      } catch (err) {
        console.error(err);
        showFieldError(textarea, 'Network error');
      } finally {
        btn.disabled = false;
      }
    });
  }

  /* ===============================
     DELETE COMMENT
  =============================== */
  function initDeleteComments() {
    const modal = document.getElementById('confirmDeleteModal');
    const confirmBtn = document.getElementById('confirmDeleteBtn');
    if (!modal || !confirmBtn) return;

    document.addEventListener('click', (e) => {
      const btn = e.target.closest('.btn-delete-comment');
      if (!btn) return;

      confirmBtn.dataset.deleteUrl = btn.dataset.deleteUrl;
      confirmBtn.dataset.itemId = btn.dataset.itemId;
    });

    confirmBtn.addEventListener('click', async () => {
      const url = confirmBtn.dataset.deleteUrl;
      const itemId = confirmBtn.dataset.itemId;
      if (!url) return;

      confirmBtn.disabled = true;

      try {
        const resp = await fetch(url, {
          method: 'POST',
          headers: {
            'X-CSRFToken': CSRF,
            'X-Requested-With': 'XMLHttpRequest',
            'Accept': 'application/json'
          }
        });

        const data = await parseJsonSafe(resp);

        if (resp.ok && data?.success) {
          document.getElementById('comment-' + data.comment_id)?.remove();

          const count = data.comments_count;

          updateDetailCounter(count);
          updateCardCounter(itemId, count);
          updateListingComments(itemId, count);

// static/js/reply_comment.js
(function () {
  'use strict';

  function clearReplyError(container) {
    const errorsBox = container.querySelector('.reply-form-errors');
    if (errorsBox) errorsBox.textContent = '';
  }

  /* ===============================
     SHOW REPLY FORM
  =============================== */

  document.addEventListener('click', function (e) {
    const btn = e.target.closest('.btn-reply');
    if (!btn) return;

    e.preventDefault();

    const commentId = btn.dataset.commentId;
    const username = btn.dataset.username;

    const container = document.getElementById('reply-form-' + commentId);
    if (!container) return;

    // скрыть все остальные reply формы
    document.querySelectorAll('.reply-form').forEach(el => {
      el.classList.add('d-none');
    });

    container.classList.remove('d-none');

    // 🔥 ОЧИЩАЕМ СТАРУЮ ОШИБКУ
    clearReplyError(container);

    const textarea = container.querySelector('textarea[name="text"]');
    if (!textarea) return;

    textarea.dataset.mention = username;
    textarea.value = '';
    textarea.focus();

    textarea.addEventListener('input', () => {
      clearReplyError(container);
    });

    // 🔥 подсветка активного comment-блока
    document.querySelectorAll('.comment-active').forEach(el => {
      el.classList.remove('comment-active');
    });

    const commentBlock = document.getElementById('comment-' + commentId);
    if (commentBlock) {
      commentBlock.classList.add('comment-active');
    }
  });

  /* ===============================
     CANCEL REPLY
  =============================== */

  document.addEventListener('click', function (e) {
    const btn = e.target.closest('.cancel-reply');
    if (!btn) return;

    const form = btn.closest('.reply-form');
    if (form) {
      clearReplyError(form);
      const textarea = form.querySelector('textarea[name="text"]');
      if (textarea) {
        textarea.value = '';
        delete textarea.dataset.mention;
      }
      form.classList.add('d-none');
    }

    document.querySelectorAll('.comment-active').forEach(el => {
      el.classList.remove('comment-active');
    });
  });

  /* ===============================
   SUBMIT REPLY
=============================== */

  document.addEventListener('submit', async function (e) {
    const form = e.target.closest('.reply-comment-form');
    if (!form) return;

    e.preventDefault();

    const textarea = form.querySelector('textarea[name="text"]');
    const errorsBox = form.querySelector('.reply-form-errors');

    if (errorsBox) errorsBox.innerHTML = '';

    const text = textarea.value.trim();
    const mention = textarea.dataset.mention;

    if (!text) {
      errorsBox.textContent = 'Please write a reply *';
      textarea.focus();
      return;
    }

    const parentId = form.dataset.parentId;
    const parentComment = form.closest('[id^="comment-"]');
    if (!parentComment) return;

    const url = parentComment.dataset.replyUrl;
    if (!url) return;

    let finalText = text;
    if (mention) {
      finalText = `@${mention}, ${text}`;
    }

    const fd = new FormData();
    fd.append('text', finalText);
    fd.append('parent_id', parentId);

    const csrf = document.querySelector('[name=csrfmiddlewaretoken]')?.value;

    try {
      const resp = await fetch(url, {
        method: 'POST',
        headers: {
          'X-CSRFToken': csrf,
          'X-Requested-With': 'XMLHttpRequest'
        },
        body: fd
      });

      const data = await resp.json().catch(() => null);
      if (!resp.ok || !data?.success) return;

      let repliesBlock = parentComment.querySelector('.replies');

      if (!repliesBlock) {
        repliesBlock = document.createElement('div');
        repliesBlock.className = 'replies mt-3';
        parentComment.appendChild(repliesBlock);
      }

      repliesBlock.insertAdjacentHTML('afterbegin', data.comment_html);

      // 🔥 помечаем только что добавленный reply
      const newReply = repliesBlock.firstElementChild;
      if (newReply) {
        newReply.classList.add('is-new-reply');

        // убрать класс после анимации
        newReply.addEventListener('animationend', () => {
          newReply.classList.remove('is-new-reply');
        }, { once: true });
      }

      // 🔥 ПОЛНАЯ инициализация reply
      restoreCommentUI(repliesBlock);
      document.addEventListener('DOMContentLoaded', () => {
        linkMentions();
      });

      // ✅ ДОБАВЛЯЕМ BADGE "Replied", ЕСЛИ ЕГО ЕЩЁ НЕТ
      let repliedBadge = parentComment.querySelector('.replied-badge');

      if (!repliedBadge) {
        // верхняя строка с автором
        const headerRow = parentComment.querySelector(
          '.d-flex.justify-content-between.align-items-center'
        );

        if (headerRow) {
          const badge = document.createElement('span');
          badge.className = 'custom_badge_success replied-badge';
          badge.innerHTML = '<i class="fa fa-reply me-2"></i>Replied';

          // вставляем рядом с именем автора
          const authorBlock = headerRow.querySelector('.d-flex.gap-1.align-items-center');
          if (authorBlock) {
            authorBlock.appendChild(badge);
          }
        }

      }

      // cleanup
      textarea.value = '';
      delete textarea.dataset.mention;
      form.closest('.reply-form').classList.add('d-none');
      parentComment.classList.remove('comment-active');

    } catch (err) {
      console.error(err);
      errorsBox.textContent = 'Network error.';
    }

  });

  document.addEventListener('DOMContentLoaded', () => {
    linkMentions(document);
  });
})();



/* ===============================
   MENTION → LINK (@username)
=============================== */
function linkMentions(container = document) {
  container.querySelectorAll('.comment-text').forEach(el => {
    if (el.dataset.mentionsLinked) return;

    const parentId = el.dataset.parentId;
    if (!parentId) return;

    el.innerHTML = el.innerHTML.replace(
      /^@([a-zA-Z0-9._-]+),?/,
      `<a href="#comment-${parentId}"
          class="mention-link"
          data-parent-id="${parentId}">
        @$1
       </a>,`
    );

    el.dataset.mentionsLinked = '1';
  });
}

function restoreCommentUI(scope) {
  if (!scope) return;

  // 1. Сбрасываем toggle-флаги
  scope.querySelectorAll('.comment-text').forEach(el => {
    delete el.dataset.toggleInit;
    delete el.dataset.mentionsLinked;
  });

  // 2. Удаляем старые кнопки
  scope.querySelectorAll('.comment-toggle-btn').forEach(btn => btn.remove());

  // 3. Пересобираем toggle
  if (window.initCommentToggles) {
    initCommentToggles(scope);
  }

  // 4. Перелинковываем mention
  linkMentions(scope);
}

function isInViewport(el, offset = 0) {
  const rect = el.getBoundingClientRect();
  const vh = window.innerHeight || document.documentElement.clientHeight;

  return (
    rect.top >= offset &&
    rect.top <= vh - offset
  );
}
