    const ta = document.querySelector('#id_text_for_ckeditor'); // твой id
    if (!ta) return;

    ClassicEditor
        .create(ta, {
            toolbar: {
                 items: [
                    'heading', '|',
                    'bold', 'italic', 'underline', 'blockQuote', 'alignment', '|',
                    'link', 'bulletedList', 'numberedList', '|',
                    'insertTable', 'undo', 'redo'
                ]
            },
            heading: {
                options: [
                    { model: 'paragraph', title: 'Paragraph', class: 'ck-heading_paragraph' },
                    { model: 'heading1', view: 'h1', title: 'Heading 1' },
                    { model: 'heading2', view: 'h2', title: 'Heading 2' }
                ]
            }
        })
        .then(editor => {
            // задаём высоту редактора
            editor.ui.view.editable.element.style.minHeight = '420px';
            window.__itemEditor = editor;
            // при сабмите формы записываем данные в textarea (если нужно)
            const form = ta.closest('form');
            if (form) form.addEventListener('submit', () => editor.updateSourceElement && editor.updateSourceElement());
        })
        .catch(err => console.error('CKEditor init error', err));
});

// static/js/clear_btn.js
document.addEventListener('DOMContentLoaded', function () {
    'use strict';

    const wrapper = document.querySelector('.search-wrapper');
    // Если нет ни одного search-wrapper на странице — выходим
    if (!wrapper) return;

    const input = wrapper.querySelector('.search-input');
    const clearBtn = wrapper.querySelector('.search-clear');

    if (!input || !clearBtn) return;

    // Инициализация: показать/скрыть в зависимости от содержимого (на случай restore/cached pages)
    clearBtn.style.opacity = input.value.trim() ? '1' : '0';

    input.addEventListener('input', function () {
        clearBtn.style.opacity = input.value.trim() ? '1' : '0';
    });

    clearBtn.addEventListener('click', function (e) {
        e.preventDefault();
        input.value = '';
        clearBtn.style.opacity = '0';
        input.focus();
        // если используешь форму — можно дополнительно триггернуть input событие:
        input.dispatchEvent(new Event('input', { bubbles: true }));
    });
});
